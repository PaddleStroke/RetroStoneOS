# RetroStoneOS for the Raspberry Pi 2 (and the Pi 3 / Zero 2 W in 32-bit
# mode): firmware configuration. Community-tested port (docs/boards.md): not
# tested on hardware by the RetroStoneOS developers.
# https://www.raspberrypi.com/documentation/computers/config_txt.html

start_file=start.elf
fixup_file=fixup.dat
kernel=zImage

# The frontend drives the display with DRM/KMS: the vc4 driver does both the
# HDMI output and the 3D (Mesa vc4, GLES 2.0 for the N64 cores) on the
# Pi 0-3. Its buffers come from CMA (the overlay's default size).
dtoverlay=vc4-kms-v3d
max_framebuffers=2
disable_overscan=1

# Faster boot: no rainbow splash, no boot delay.
disable_splash=1
boot_delay=0

# Serial console on GPIO 14/15. The Pi 2 has only the PL011 there
# (ttyAMA0). On a Pi 3 / Zero 2 W the PL011 normally drives Bluetooth:
# miniuart-bt moves Bluetooth to the mini UART, so the console is ttyAMA0 on
# every model this image boots (board.ini serial_console).
enable_uart=1
[pi3]
dtoverlay=miniuart-bt
[pi02]
dtoverlay=miniuart-bt
[all]

# Sound goes to the TV (vc4 HDMI audio); the 3.5 mm jack is not used (the
# frontend's audio follows the display, which is always HDMI here).
dtparam=audio=off
