# RetroStoneOS for the Raspberry Pi 4 (64-bit): firmware configuration.
# Community-tested reference port (docs/porting.md): not tested on hardware
# by the RetroStoneOS developers.
# https://www.raspberrypi.com/documentation/computers/config_txt.html

start_file=start4.elf
fixup_file=fixup4.dat
kernel=Image
arm_64bit=1

# The frontend drives the display with DRM/KMS: the vc4 KMS driver (HDMI)
# and the V3D GPU (Mesa v3d, GLES 2.0 for the N64 cores).
dtoverlay=vc4-kms-v3d
max_framebuffers=2
disable_overscan=1

# Faster boot: no rainbow splash, no boot delay.
disable_splash=1
boot_delay=0

# Serial console on GPIO 14/15 (mini UART, ttyS0; Bluetooth keeps the PL011).
enable_uart=1

# Sound goes to the TV (vc4 HDMI audio); the 3.5 mm jack is not used.
dtparam=audio=off
